`timescale 1ns / 1ps

module MasterSelect(
    input clk,
    input reset,
    input sw0,
    input sw1,
    output reg [1:0] mode_out,
    output reg [2:0] set_pos_out
);

//////////////////////////////////////////////////////////////
//  debounce parameter
//////////////////////////////////////////////////////////////
parameter DEBOUNCE = 5_000_000;

//////////////////////////////////////////////////////////////
//  sw0 debounce
//////////////////////////////////////////////////////////////
reg [1:0] sw0_sync;
reg [22:0] sw0_cnt;
reg sw0_deb, sw0_prev;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        sw0_sync <= 0;
        sw0_cnt  <= 0;
        sw0_deb  <= 0;
    end else begin
        sw0_sync <= {sw0_sync[0], sw0};

        if(sw0_sync[1] != sw0_deb) begin
            if(sw0_cnt < DEBOUNCE)
                sw0_cnt <= sw0_cnt + 1;
            else begin
                sw0_deb <= sw0_sync[1];
                sw0_cnt <= 0;
            end
        end else
            sw0_cnt <= 0;
    end
end

always @(posedge clk or posedge reset) begin
    if(reset)
        sw0_prev <= 0;
    else
        sw0_prev <= sw0_deb;
end

wire sw0_pulse = sw0_deb & ~sw0_prev;

//////////////////////////////////////////////////////////////
//  sw1 debounce
//////////////////////////////////////////////////////////////
reg [1:0] sw1_sync;
reg [22:0] sw1_cnt;
reg sw1_deb, sw1_prev;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        sw1_sync <= 0;
        sw1_cnt  <= 0;
        sw1_deb  <= 0;
    end else begin
        sw1_sync <= {sw1_sync[0], sw1};

        if(sw1_sync[1] != sw1_deb) begin
            if(sw1_cnt < DEBOUNCE)
                sw1_cnt <= sw1_cnt + 1;
            else begin
                sw1_deb <= sw1_sync[1];
                sw1_cnt <= 0;
            end
        end else
            sw1_cnt <= 0;
    end
end

always @(posedge clk or posedge reset) begin
    if(reset)
        sw1_prev <= 0;
    else
        sw1_prev <= sw1_deb;
end

wire sw1_pulse = sw1_deb & ~sw1_prev;

//////////////////////////////////////////////////////////////
//  mode 변경
//////////////////////////////////////////////////////////////
always @(posedge clk or posedge reset) begin
    if(reset)
        mode_out <= 2'b00;
    else if(sw0_pulse)
        mode_out <= mode_out + 1;
end

//////////////////////////////////////////////////////////////
//  set position 변경
//////////////////////////////////////////////////////////////
always @(posedge clk or posedge reset) begin
    if(reset)
        set_pos_out <= 3'b100;
    else if(sw1_pulse && (mode_out == 2'b01 || mode_out == 2'b11)) begin
        case(set_pos_out)
            3'b100: set_pos_out <= 3'b010;
            3'b010: set_pos_out <= 3'b001;
            3'b001: set_pos_out <= 3'b100;
            default: set_pos_out <= 3'b100;
        endcase
    end
end

endmodule

module FNDDisplay(
    input [1:0] mode,
    input [5:0] sec_in, min_in,
    input [4:0] hour_in,
    input [4:0] al_hour,
    input [5:0] al_min, al_sec,
    input [6:0] tl_mmsec,
    input [5:0] tl_sec, tl_min,
    output reg[3:0] hour10, hour0,
    output reg[3:0] min10, min0,
    output reg[3:0] sec10, sec0
    );
always @(*) begin
    hour10 = 0; hour0 = 0;
    min10 = 0; min0 = 0;
    sec10 = 0; sec0 = 0;
    case(mode)
    2'b00, 2'b01: begin
        hour10 = hour_in / 10;  hour0 = hour_in % 10;
        min10 = min_in / 10;    min0 = min_in % 10;
        sec10 = sec_in / 10;    sec0 = sec_in % 10;
     end
     2'b10 : begin
        hour10 = tl_min / 10;   hour0 = tl_min % 10;
        min10 = tl_sec / 10;    min0 = tl_sec % 10;
        sec10 = tl_mmsec / 10;  sec0 = tl_mmsec % 10;
     end
     2'b11 : begin
        hour10 = al_hour / 10;  hour0 = al_hour % 10;
        min10 = al_min / 10;    min0 = al_min % 10;
        sec10 = al_sec / 10;    sec0 = al_sec % 10;
     end
     endcase
     end
endmodule

module clockdivider_paramN #(parameter N = 100_000)
(
    input clk,
    input reset,
    output reg clk_out
);

    reg [31:0] cnt;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        cnt <= 0;
        clk_out <= 0;
    end else begin
        if(cnt == (N/2 -1)) begin
            cnt <= 0;
            clk_out <= ~clk_out;   // ⭐ 토글!
        end else begin
            cnt <= cnt + 1;
        end
    end
end

endmodule

module datamux(
    input [3:0] ina, inb, inc, ind, ine, inf,
    input [1:0] insel,
    input time_mode,
    output reg [3:0] outy
);
    always @(insel or time_mode)
        begin
        if(time_mode == 1'b0) begin
            case(insel)
                2'b00 : outy = ina;
                2'b01 : outy = inb;
                2'b10 : outy = inc;
                2'b11 : outy = ind;
            endcase
            end
        else begin
            case(insel)
                2'b00 : outy =inc;
                2'b01 : outy =ind;
                2'b10 : outy =ine;
                2'b11 : outy =inf;
            endcase
        end    
     end           
endmodule

        4'h7 : fnd = 7'b0001111;
        4'h8 : fnd = 7'b0000000;
        4'h9 : fnd = 7'b0000100;
        default : fnd = 7'b1111111;
    endcase
end

module Timer(
    input clk,
    input reset,
    input [1:0] mode,
    input clk1000Hz,
    input sw1,
    input sw2,
    output reg [6:0] mmsec_out,
    output reg [5:0] sec_out,
    output reg [5:0] min_out
);

wire run = sw1;
wire rst = sw2;

always @(posedge clk1000Hz or posedge reset) begin
    if(reset || (mode == 2'b10 && rst)) begin
        mmsec_out <= 0;
        sec_out <= 0;
        min_out <= 0;
    end
    else if(mode == 2'b10 && run) begin
        if(mmsec_out == 99) begin
            mmsec_out <= 0;
            if(sec_out == 59) begin
                sec_out <= 0;
                if(min_out == 59)
                    min_out <= 0;
                else
                    min_out <= min_out + 1;
            end else
                sec_out <= sec_out + 1;
        end else
            mmsec_out <= mmsec_out + 1;
    end
end

endmodule



`timescale 1ns / 1ps

module AlarmLogic(
    input clk,
    input reset,
    input clk1000hz,
    input [1:0] mode,
    input [2:0] set_pos,
    input [5:0] clk_min,
    input [4:0] clk_hour,
    input sw2,
    input sw3,

    output [5:0] sec_out,
    output [5:0] min_out,
    output [4:0] hour_out,
    output reg [2:0] alarm_out,
    output reg [2:0] alarm_on
);

//////////////////////////////////////////////////////////////
//  sw2 debounce (시간 설정 버튼)
//////////////////////////////////////////////////////////////
reg [1:0] sw2_sync;
reg [19:0] sw2_cnt;
reg sw2_deb, sw2_prev;

parameter DEBOUNCE = 5_000_000;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        sw2_sync <= 0;
        sw2_cnt <= 0;
        sw2_deb <= 0;
    end else begin
        sw2_sync <= {sw2_sync[0], sw2};

        if(sw2_sync[1] != sw2_deb) begin
            if(sw2_cnt < DEBOUNCE)
                sw2_cnt <= sw2_cnt + 1;
            else begin
                sw2_deb <= sw2_sync[1];
                sw2_cnt <= 0;
            end
        end else
            sw2_cnt <= 0;
    end
end

always @(posedge clk or posedge reset) begin
    if(reset)
        sw2_prev <= 0;
    else
        sw2_prev <= sw2_deb;
end

wire sw2_pulse = sw2_deb & ~sw2_prev;


//////////////////////////////////////////////////////////////
//  sw3 debounce (알람 ON/OFF)
//////////////////////////////////////////////////////////////
reg [1:0] sw3_sync;
reg [19:0] sw3_cnt;
reg sw3_deb, sw3_prev;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        sw3_sync <= 0;
        sw3_cnt <= 0;
        sw3_deb <= 0;
    end else begin
        sw3_sync <= {sw3_sync[0], sw3};

        if(sw3_sync[1] != sw3_deb) begin
            if(sw3_cnt < DEBOUNCE)
                sw3_cnt <= sw3_cnt + 1;
            else begin
                sw3_deb <= sw3_sync[1];
                sw3_cnt <= 0;
            end
        end else
            sw3_cnt <= 0;
    end
end

always @(posedge clk or posedge reset) begin
    if(reset)
        sw3_prev <= 0;
    else
        sw3_prev <= sw3_deb;
end

wire sw3_pulse = sw3_deb & ~sw3_prev;


//////////////////////////////////////////////////////////////
// 🔥 알람 ON/OFF 토글
//////////////////////////////////////////////////////////////
reg alarm;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        alarm <= 0;
        alarm_on <= 3'b000;
    end 
    else if(sw3_pulse) begin
        alarm <= ~alarm;
        alarm_on <= (~alarm) ? 3'b111 : 3'b000;
    end
end


//////////////////////////////////////////////////////////////
// 🔥 알람 시간 레지스터
//////////////////////////////////////////////////////////////
reg [5:0] sec, min;
reg [4:0] hour;

always @(posedge clk or posedge reset) begin
    if(reset) begin
        sec <= 0;
        min <= 0;
        hour <= 0;
    end
    else if(mode == 2'b11 && sw2_pulse) begin
        case(set_pos)
            3'b001: sec  <= (sec==59)?0:sec+1;
            3'b010: min  <= (min==59)?0:min+1;
            3'b100: hour <= (hour==23)?0:hour+1;
        endcase
    end
end

assign sec_out = sec;
assign min_out = min;
assign hour_out = hour;


//////////////////////////////////////////////////////////////
//  알람 비교 (1000Hz 기준)
//////////////////////////////////////////////////////////////
always @(posedge clk1000hz or posedge reset) begin
    if(reset)
        alarm_out <= 3'b000;
    else if((hour == clk_hour) && (min == clk_min) && alarm && (mode != 2'b11))
        alarm_out <= 3'b111;
    else
        alarm_out <= 3'b000;
end

endmodule
